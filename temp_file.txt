"""
Enhanced Agent Lifecycle System for AetherOS Orchestration
"""

from __future__ import annotations

from enum import Enum
from typing import Optional
import time


class AgentLifecycleState(Enum):
    """Complete agent lifecycle states matching the master prompt specification"""
    CREATED = "created"
    INITIALIZING = "initializing"
    READY = "ready"
    RUNNING = "running"
    WAITING = "waiting"
    PAUSED = "paused"
    BLOCKED = "blocked"
    VERIFYING = "verifying"
    COMPLETED = "completed"
    FAILED = "failed"
    RECOVERING = "recovering"
    TERMINATED = "terminated"
    QUARANTINED = "quarantined"


class LifecycleTransition:
    """Defines valid lifecycle transitions"""
    
    # Define valid transitions for each state
    VALID_TRANSITIONS = {
        AgentLifecycleState.CREATED: [
            AgentLifecycleState.INITIALIZING,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.INITIALIZING: [
            AgentLifecycleState.READY,
            AgentLifecycleState.FAILED,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.READY: [
            AgentLifecycleState.RUNNING,
            AgentLifecycleState.PAUSED,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.RUNNING: [
            AgentLifecycleState.WAITING,
            AgentLifecycleState.PAUSED,
            AgentLifecycleState.BLOCKED,
            AgentLifecycleState.VERIFYING,
            AgentLifecycleState.COMPLETED,
            AgentLifecycleState.FAILED,
            AgentLifecycleState.RECOVERING,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.WAITING: [
            AgentLifecycleState.RUNNING,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.PAUSED: [
            AgentLifecycleState.READY,
            AgentLifecycleState.RUNNING,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.BLOCKED: [
            AgentLifecycleState.RUNNING,
            AgentLifecycleState.RECOVERING,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.VERIFYING: [
            AgentLifecycleState.RUNNING,
            AgentLifecycleState.COMPLETED,
            AgentLifecycleState.FAILED,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.COMPLETED: [
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.FAILED: [
            AgentLifecycleState.RECOVERING,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.RECOVERING: [
            AgentLifecycleState.RUNNING,
            AgentLifecycleState.FAILED,
            AgentLifecycleState.TERMINATED
        ],
        AgentLifecycleState.TERMINATED: [],  # Terminal state
        AgentLifecycleState.QUARANTINED: [
            AgentLifecycleState.TERMINATED
        ]
    }
    
    @staticmethod
    def is_valid_transition(from_state: AgentLifecycleState, to_state: AgentLifecycleState) -> bool:
        """Check if a transition is valid"""
        if from_state not in LifecycleTransition.VALID_TRANSITIONS:
            return False
        return to_state in LifecycleTransition.VALID_TRANSITIONS[from_state]
    
    @staticmethod
    def get_valid_transitions(state: AgentLifecycleState) -> list[AgentLifecycleState]:
        """Get all valid transitions from a state"""
        return LifecycleTransition.VALID_TRANSITIONS.get(state, [])


class LifecycleEvent:
    "Represents a lifecycle event"

    def __init__(self, agent_id: str, from_state: AgentLifecycleState,
                 to_state: AgentLifecycleState, timestamp: float = None,
                 metadata: dict = None):
        self.agent_id = agent_id
        self.from_state = from_state
        self.to_state = to_state
        self.timestamp = timestamp or time.time()
        self.metadata = metadata or {}

    def to_dict(self):
        return {
            "agent_id": self.agent_id,
            "from_state": self.from_state.value,
            "to_state": self.to_state.value,
            "timestamp": self.timestamp,
            "metadata": self.metadata
        }
class LifecycleTracker:
    """Tracks lifecycle events for an agent"""
    
    def __init__(self, agent_id: str):
        self.agent_id = agent_id
        self._history: list[LifecycleEvent] = []
        self._current_state: AgentLifecycleState = AgentLifecycleState.CREATED
    
    def transition(self, to_state: AgentLifecycleState, 
                   metadata: dict = None) -> bool:
        """Transition to a new state if valid"""
        if not LifecycleTransition.is_valid_transition(self._current_state, to_state):
            return False
        
        from_state = self._current_state
        self._current_state = to_state
        
        event = LifecycleEvent(
            self.agent_id, 
            from_state, 
            to_state, 
            metadata=metadata
        )
        self._history.append(event)
        return True
    
    def get_current_state(self) -> AgentLifecycleState:
        """Get current lifecycle state"""
        return self._current_state
    
    def get_history(self) -> list[LifecycleEvent]:
        """Get lifecycle event history"""
        return self._history.copy()
    
    def get_state_duration(self, state: AgentLifecycleState) -> float:
        """Get total time spent in a specific state"""
        total = 0.0
        last_time = None
        
        for event in self._history:
            if event.from_state == state:
                if last_time is not None:
                    total += event.timestamp - last_time
                last_time = event.timestamp
            elif event.to_state == state:
                last_time = event.timestamp
        
        # If currently in the state, add time from last transition
        if last_time is not None and self._current_state == state:
            total += time.time() - last_time
            
        return total
    
    def to_dict(self):
        return {
            "agent_id": self.agent_id,
            "current_state": self._current_state.value,
            "history": [event.to_dict() for event in self._history]
        }
# Global lifecycle registry
class LifecycleRegistry:
    """Registry for tracking all agent lifecycles"""
    
    def __init__(self):
        self._trackers: dict[str, LifecycleTracker] = {}
        self._lock = __import__('threading').RLock()
    
    def ensure(self, agent_id: str) -> LifecycleTracker:
        """Get or create a lifecycle tracker for an agent"""
        with self._lock:
            if agent_id not in self._trackers:
                self._trackers[agent_id] = LifecycleTracker(agent_id)
            return self._trackers[agent_id]
    
    def get(self, agent_id: str) -> Optional[LifecycleTracker]:
        """Get lifecycle tracker for an agent"""
        with self._lock:
            return self._trackers.get(agent_id)
    
    def remove(self, agent_id: str) -> bool:
        """Remove lifecycle tracker for an agent"""
        with self._lock:
            if agent_id in self._trackers:
                del self._trackers[agent_id]
                return True
            return False
    
    def get_state_distribution(self) -> dict[AgentLifecycleState, int]:
        """Get count of agents in each state"""
        with self._lock:
            distribution = {}
            for tracker in self._trackers.values():
                state = tracker.get_current_state()
                distribution[state] = distribution.get(state, 0) + 1
            return distribution
    
    def list_agents_in_state(self, state: AgentLifecycleState) -> list[str]:
        """List all agent IDs in a specific state"""
        with self._lock:
            return [
                agent_id for agent_id, tracker in self._trackers.items()
                if tracker.get_current_state() == state
            ]


# Global instance
lifecycle_registry = LifecycleRegistry()
